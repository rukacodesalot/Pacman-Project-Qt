void MainWindow::moveEnemy() {
    enemyPos.rx() += enemyDx;
    enemyPos.ry() += enemyDy;


    if (enemyPos.x() <= 0 || enemyPos.x() >= width() - 30) {
        enemyDx = -enemyDx;
    }

void MainWindow::gameLoop() {
    pacman->move(walls);
    moveEnemy();


    QRect pacmanRect(pacman->getX(), pacman->getY(), 30, 30);


    for (int i = 0; i < pellets.size(); ++i) {
        QRect pelletRect(pellets[i].x(), pellets[i].y(), 10, 10);
        if (pacmanRect.intersects(pelletRect)) {
            pellets.remove(i);
            score++;
            break;
        }
        if (pellets.isEmpty()) {
            if (level >= 3) { // or any number you want
                timer->stop();
                QMessageBox::information(this, "You Win!", "Congratulations, you cleared all levels!");
                return;
            }




            initializeWalls();     // optional: change wall layout based on level
            initializePellets();   // refill pellets
            resetPositions();


            // Increase difficulty
            int interval = qMax(30, 100 - (level * 10)); // Speed up
            timer->setInterval(interval);
        }
            }


    QRect enemyRect(enemyPos.x(), enemyPos.y(), 30, 30);
    if (pacmanRect.intersects(enemyRect)) {
        lives--;
        if (lives <= 0) {
            timer->stop();
            QMessageBox::information(this, "Game Over", "You were caught!");
        } else {
            resetPositions();
        }
    }




    update();
}


void MainWindow::paintEvent(QPaintEvent *) {

void MainWindow::keyPressEvent(QKeyEvent *event) {
    if (!gameRunning && event->key() == Qt::Key_Space) {
        gameRunning = true;
        timer->start();
        return;
    }


    if (event->key() == Qt::Key_P) {
        gameRunning = !gameRunning;
        if (gameRunning) timer->start();
        else timer->stop();
        return;
    }
    if (event->key() == Qt::Key_R) {
        score = 0;
        lives = 3;
        level = 1;
        initializeWalls();
        initializePellets();
        resetPositions();
        gameRunning = true;
        timer->start();
        return;
    }
    if (event->key() == Qt::Key_Escape) {
        close();
        return;
    }


    pacman->setDirection(event->key());
}
void MainWindow::initializePellets() {
    pellets.clear();
    for (int y = 20; y < 400; y += 20) {
        for (int x = 20; x < 400; x += 20) {
            QPoint pt(x, y);
            QRect pelletRect(pt.x(), pt.y(), 10, 10); // pellet size


            bool isWall = false;
            for (const QPoint &wall : walls) {
                QRect wallRect(wall.x(), wall.y(), 20, 20); // wall size
                if (pelletRect.intersects(wallRect)) {
                    isWall = true;
                    break;
                }
            }


            if (!isWall) {
                pellets.append(pt);
            }
        }
    }
}

// Constructor to position the pacman in its initial position
Pacman::Pacman(QWidget *parent)
    : QWidget(parent), x(60), y(60), dx(0), dy(0) {}


// Set direction based on key press
void Pacman::setDirection(int key) {
    switch (key) {
    case Qt::Key_Up:
        dx = 0;
        dy = -10; // Move up (negative y direction)
        break;
    case Qt::Key_Down:
        dx = 0;
        dy = 10; // Move down (positive y direction)
        break;
    case Qt::Key_Left:
        dx = -10;
        dy = 0; // Move left (negative x direction)
        break;
    case Qt::Key_Right:
        dx = 10;
        dy = 0; // Move right (positive x direction)
        break;
    default:
        break;
    }
}


// Update Pacman's position
void Pacman::move(const QVector<QPoint> &walls) {
    int newX = x + dx;
    int newY = y + dy;
    QRect nextRect(newX+5, newY+5, 20, 20);


    for (const QPoint &wall : walls) {
        QRect wallRect(wall.x(), wall.y(), 20, 20);
        if (nextRect.intersects(wallRect)) {
            return; // Blocked by wall
        }
    }


    x = newX;
    y = newY;


    if (x < 0) x = 0;
    if (x > 370) x = 370;
    if (y < 0) y = 0;
    if (y > 370) y = 370;
}




// Draw Pacman using an image
void Pacman::draw(QPainter *painter) {
    QPixmap pacmanImg(":/images/pics/pacman.png");
    painter->drawPixmap(x, y, 20, 20, pacmanImg);
}
void Pacman::setPos(int newX, int newY) {
    x = newX;
    y = newY;
}
